  PROJETO 1 — CARD DE PERFIL (Básico)
  Conceitos: Box model, border-radius, box-shadow, Flexbox (centralizar)

<!DOCTYPE html>
<html lang="pt-br">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Projeto 1 — Card de Perfil</title>
    <style>
        /* ══════════════════════════════════
           RESET — Remove estilos padrão do navegador.
           Sempre comece com isso!
           ══════════════════════════════════ */
        * {
            margin: 0;
            padding: 0;
            box-sizing: border-box;  /* width inclui padding e border */
        }

        /* ══════════════════════════════════
           BODY — Centraliza o card na tela
           usando Flexbox.
           ══════════════════════════════════ */
        body {
            font-family: 'Segoe UI', Arial, sans-serif;
            background-color: #f0f2f5;
            min-height: 100vh;         /* 100% da altura da janela */

            /* FLEXBOX para centralizar vertical + horizontal */
            display: flex;
            justify-content: center;   /* centraliza horizontalmente */
            align-items: center;       /* centraliza verticalmente */
        }

        /* ══════════════════════════════════
           CARD — O container principal.
           Demonstra: width, padding, border-radius,
           box-shadow, text-align.
           ══════════════════════════════════ */
        .card {
            background: white;
            width: 320px;              /* largura fixa */
            padding: 2rem;             /* respiro interno (32px) */
            border-radius: 16px;       /* cantos arredondados */
            text-align: center;        /* centraliza o conteúdo */

            /* SOMBRA: horizontal vertical blur spread cor */
            box-shadow: 0 8px 24px rgba(0, 0, 0, 0.12);
        }

        /* ══════════════════════════════════
           FOTO DO PERFIL — Círculo perfeito.
           border-radius: 50% transforma qualquer
           elemento quadrado em um círculo.
           ══════════════════════════════════ */
        .foto {
            width: 120px;
            height: 120px;
            border-radius: 50%;        /* CÍRCULO! */
            margin: 0 auto 1rem;       /* centraliza e adiciona espaço abaixo */

            /* Borda decorativa ao redor da foto */
            border: 4px solid #3498db;

            /* Imagem de exemplo (substituir pela sua) */
            background-image: url('https://i.pravatar.cc/120');
            background-size: cover;    /* preenche todo o espaço */
            background-position: center;
        }

        /* ══════════════════════════════════
           TEXTOS DO CARD
           ══════════════════════════════════ */
        .nome {
            font-size: 1.4rem;
            font-weight: 700;
            color: #2c3e50;
            margin-bottom: 0.3rem;
        }

        .cargo {
            font-size: 0.9rem;
            color: #7f8c8d;            /* cinza mais claro */
            margin-bottom: 1.5rem;
        }

        /* ══════════════════════════════════
           INFORMAÇÕES — Pequena lista de dados.
           Usa border-top para separar visualmente.
           ══════════════════════════════════ */
        .info {
            display: flex;             /* itens lado a lado */
            justify-content: space-around; /* distribui igualmente */
            border-top: 1px solid #eee;
            padding-top: 1rem;
            margin-bottom: 1.5rem;
        }

        .info-item {
            text-align: center;
        }

        .info-numero {
            font-size: 1.2rem;
            font-weight: 700;
            color: #2c3e50;
        }

        .info-label {
            font-size: 0.75rem;
            color: #95a5a6;
            text-transform: uppercase;  /* TUDO MAIÚSCULO */
            letter-spacing: 1px;
        }

        /* ══════════════════════════════════
           BOTÃO — Demonstra transition para
           animação suave no hover.
           ══════════════════════════════════ */
        .botao {
            display: inline-block;
            background-color: #3498db;
            color: white;
            padding: 10px 32px;
            border: none;
            border-radius: 25px;       /* muito arredondado = "pílula" */
            font-size: 0.9rem;
            font-weight: 600;
            text-decoration: none;     /* remove sublinhado (é um <a>) */
            cursor: pointer;

            /* TRANSIÇÃO: anima todas as propriedades por 0.3s */
            transition: all 0.3s ease;
        }

        /* Estado hover: quando o mouse passa por cima */
        .botao:hover {
            background-color: #2980b9; /* cor mais escura */
            transform: translateY(-2px); /* sobe 2px */
            box-shadow: 0 4px 12px rgba(52, 152, 219, 0.4);
        }
    </style>

  PROJETO 2 — MENU DE NAVEGAÇÃO RESPONSIVO (Básico)
  Conceitos: Flexbox, media query, :hover, transições, list-style
================================================================================

<!DOCTYPE html>
<html lang="pt-br">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Projeto 2 — Menu Responsivo</title>
    <style>
        * {
            margin: 0;
            padding: 0;
            box-sizing: border-box;
        }

        body {
            font-family: 'Segoe UI', Arial, sans-serif;
            background-color: #f4f4f4;
        }

        /* ══════════════════════════════════
           HEADER — Barra de navegação fixa no topo.
           position: fixed faz o header "flutuar"
           sobre o conteúdo ao rolar a página.
           ══════════════════════════════════ */
        header {
            background-color: #2c3e50;
            color: white;
            padding: 0 2rem;
            position: fixed;           /* fixa no topo */
            top: 0;
            left: 0;
            right: 0;                  /* estica de ponta a ponta */
            z-index: 100;              /* fica acima de tudo */

            /* FLEXBOX: logo à esquerda, menu à direita */
            display: flex;
            justify-content: space-between;
            align-items: center;
            height: 60px;
        }

        /* ══════════════════════════════════
           LOGO
           ══════════════════════════════════ */
        .logo {
            font-size: 1.3rem;
            font-weight: 700;
            letter-spacing: 1px;
        }

        .logo span {
            color: #3498db;            /* destaque na parte do nome */
        }

        /* ══════════════════════════════════
           MENU — Lista horizontal com Flexbox.
           list-style: none remove os bullets.
           ══════════════════════════════════ */
        nav ul {
            list-style: none;          /* remove bullets */
            display: flex;             /* itens lado a lado */
            gap: 0;                    /* sem gap — usamos padding nos links */
        }

        nav a {
            color: rgba(255, 255, 255, 0.8); /* branco semitransparente */
            text-decoration: none;           /* remove sublinhado */
            padding: 20px 16px;              /* área clicável generosa */
            font-size: 0.9rem;
            font-weight: 500;
            display: block;                  /* ocupa todo o <li> */

            /* Transição suave na cor e no fundo */
            transition: color 0.3s, background-color 0.3s;
        }

        /* HOVER: ao passar o mouse */
        nav a:hover {
            color: white;
            background-color: rgba(255, 255, 255, 0.1); /* fundo translúcido */
        }

        /* Link ativo (página atual) */
        nav a.ativo {
            color: white;
            border-bottom: 3px solid #3498db; /* barra azul embaixo */
        }

        /* ══════════════════════════════════
           CONTEÚDO — Margem superior para
           não ficar escondido atrás do header fixo.
           ══════════════════════════════════ */
        main {
            margin-top: 80px;          /* header tem 60px + 20px de folga */
            max-width: 800px;
            margin-left: auto;
            margin-right: auto;
            padding: 0 1.5rem;
        }

        section {
            background: white;
            padding: 2rem;
            margin-bottom: 1.5rem;
            border-radius: 8px;
            box-shadow: 0 2px 8px rgba(0, 0, 0, 0.08);
        }

        h2 {
            color: #2c3e50;
            margin-bottom: 0.5rem;
        }

        p {
            color: #555;
            line-height: 1.7;
        }

        /* ══════════════════════════════════
           RESPONSIVO — Telas menores que 768px.
           O menu muda de HORIZONTAL para VERTICAL.
           ══════════════════════════════════ */
        @media (max-width: 768px) {

            /* Header empilha logo e menu */
            header {
                flex-direction: column;  /* empilha vertical */
                height: auto;            /* altura flexível */
                padding: 1rem;
            }

            /* Menu vira coluna */
            nav ul {
                flex-direction: column;  /* itens empilhados */
                width: 100%;
                text-align: center;
            }

            nav a {
                padding: 12px;
                border-bottom: 1px solid rgba(255, 255, 255, 0.1);
            }

            nav a.ativo {
                border-bottom: 1px solid #3498db;
            }

            /* Mais espaço no topo (header ficou maior) */
            main {
                margin-top: 220px;
            }
        }
    </style>
</head>
<body>

    <header>
        <div class="logo">Tech<span>Blog</span></div>
        <nav>
            <ul>
                <li><a href="#" class="ativo">Início</a></li>
                <li><a href="#">Artigos</a></li>
                <li><a href="#">Sobre</a></li>
                <li><a href="#">Contato</a></li>
            </ul>
        </nav>
    </header>

    <main>
        <section>
            <h2>Bem-vindo ao TechBlog</h2>
            <p>Este é um exemplo de menu de navegação responsivo. Redimensione a janela do navegador ou use o DevTools (F12 → ícone de celular) para ver o menu se transformar de horizontal para vertical.</p>
        </section>
        <section>
            <h2>Seção de Conteúdo</h2>
            <p>Observe como o header fica fixo no topo enquanto você rola a página. Isso é feito com <code>position: fixed</code> no CSS.</p>
        </section>
        <section>
            <h2>Outra Seção</h2>
            <p>O efeito de hover nos links do menu usa <code>transition</code> para suavizar a mudança de cor. Sem transition, a mudança seria instantânea e brusca.</p>
        </section>
    </main>

</body>
</html>

<!DOCTYPE html>
<html lang="pt-br">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Projeto 3 — Galeria de Imagens</title>
    <style>
        * {
            margin: 0;
            padding: 0;
            box-sizing: border-box;
        }

        body {
            font-family: 'Segoe UI', Arial, sans-serif;
            background-color: #1a1a2e;    /* fundo escuro para contraste */
            color: white;
            padding: 2rem;
        }

        /* ══════════════════════════════════
           TÍTULO DA PÁGINA
           ══════════════════════════════════ */
        h1 {
            text-align: center;
            font-size: 2rem;
            margin-bottom: 0.5rem;
        }

        .subtitulo {
            text-align: center;
            color: rgba(255, 255, 255, 0.6);
            margin-bottom: 2rem;
            font-size: 0.95rem;
        }

        /* ══════════════════════════════════
           GALERIA — Grid flexível com flex-wrap.
           Os itens se reorganizam automaticamente
           conforme o tamanho da tela.
           ══════════════════════════════════ */
        .galeria {
            display: flex;
            flex-wrap: wrap;           /* permite quebra de linha */
            gap: 12px;                 /* espaço entre as imagens */
            max-width: 1000px;
            margin: 0 auto;           /* centraliza o container */
        }

        /* ══════════════════════════════════
           ITEM DA GALERIA — Cada imagem.
           flex: 1 1 250px significa:
             1 = pode crescer
             1 = pode encolher
             250px = tamanho mínimo antes de quebrar linha
           ══════════════════════════════════ */
        .galeria-item {
            flex: 1 1 250px;
            height: 250px;
            border-radius: 12px;

            /* overflow: hidden impede que o zoom
               da imagem "vaze" para fora do card */
            overflow: hidden;

            position: relative;        /* para posicionar o overlay */
            cursor: pointer;
        }

        /* ══════════════════════════════════
           IMAGEM — Preenche todo o card.
           transition + transform criam o efeito
           de zoom suave no hover.
           ══════════════════════════════════ */
        .galeria-item img {
            width: 100%;
            height: 100%;
            object-fit: cover;         /* preenche sem distorcer */
            display: block;

            /* Animação suave de 0.4 segundos */
            transition: transform 0.4s ease, opacity 0.4s ease;
        }

        /* HOVER: zoom + leve escurecimento */
        .galeria-item:hover img {
            transform: scale(1.08);    /* aumenta 8% */
            opacity: 0.85;             /* leve escurecimento */
        }

        /* ══════════════════════════════════
           OVERLAY — Texto que aparece no hover.
           Fica invisível por padrão (opacity: 0)
           e aparece suavemente no hover.
           ══════════════════════════════════ */
        .overlay {
            position: absolute;
            bottom: 0;
            left: 0;
            right: 0;
            padding: 1rem;
            background: linear-gradient(transparent, rgba(0, 0, 0, 0.7));
            color: white;

            /* Começa invisível */
            opacity: 0;
            transform: translateY(10px); /* deslocado para baixo */
            transition: opacity 0.4s ease, transform 0.4s ease;
        }

        /* No hover do item-pai, o overlay aparece */
        .galeria-item:hover .overlay {
            opacity: 1;
            transform: translateY(0);
        }

        .overlay h3 {
            font-size: 1rem;
            margin-bottom: 0.2rem;
        }

        .overlay p {
            font-size: 0.8rem;
            opacity: 0.8;
        }

        /* ══════════════════════════════════
           RESPONSIVO — Em telas pequenas, cada
           imagem ocupa a largura toda.
           ══════════════════════════════════ */
        @media (max-width: 600px) {
            .galeria-item {
                flex: 1 1 100%;        /* uma imagem por linha */
                height: 200px;
            }
        }
    </style>
</head>
